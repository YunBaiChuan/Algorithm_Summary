**思路：树上求直径的模板题，一般是两次dfs或者两次bfs，需要写熟，我常用两次bfs。简单来说就是从图上任意节点出发，先跑一次bfs得到最远距离的节点，这个节点必然是直径的某个端点，然后再在这个节点上跑一遍bfs，得到的就是树的直径了**

**代码：**
```python
import sys
from collections import deque

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

def solve():
    n = II()

    g = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v = MII()
        g[u].append(v)
        g[v].append(u)

    def bfs(start):
        dist = [-1] * (n + 1)
        dist[start] = 0
        q = deque([start])
        while q:
            u = q.popleft()
            for v in g[u]:
                if dist[v] == -1:
                    dist[v] = dist[u] + 1
                    q.append(v)
        farthest = max(range(1, n + 1), key=lambda x: dist[x])
        return farthest, dist[farthest]

    u, _ = bfs(1)
    v, diameter = bfs(u)
    print(diameter + 1)

t = 1
# t = II()
for _ in range(t):
    solve()
```
