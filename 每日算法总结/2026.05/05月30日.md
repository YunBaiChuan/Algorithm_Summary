**思路：动态规划，二维dp。定义$dp[i][c]$表示处理完前i个字符，并且当前i位置变成c时最小代价。那么状态转移方程为：$dp[i][c] = |s[i] - c| + min(dp[i-1][d])$，其中取min的操作可以用前缀最小值进行更新，并且优化成一维滚动数组**

**代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n = II()
    s = I()

    dp = [0] * 26

    for i in range(n):
        ch = ord(s[i]) - ord('A')
        new_dp = [inf] * 26
        pre_min = inf
        for c in range(26):
            pre_min = fmin(pre_min, dp[c])
            cost = abs(ch - c)
            new_dp[c] = pre_min + cost
        dp = new_dp

    ans = min(dp)
    print(ans)

t = 1
# t = II()
for _ in range(t):
    solve()
```
