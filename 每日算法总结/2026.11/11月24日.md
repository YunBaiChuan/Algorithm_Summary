**思路：状压dp。从左往右，从上往下进行记忆化搜索，这样的话只需要记录上一行的状态就可以进行正确的状态转移了**

import sys
from functools import lru_cache

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    n, k = MII()

    valid = [s for s in range(1 << n) if (s & (s << 1)) == 0]
    pc = [bin(s).count('1') for s in range(1 << n)]

    @lru_cache(maxsize=None)
    def dfs(row, prev_state, cnt):
        if cnt > k:
            return 0
        if row == n:
            return 1 if cnt == k else 0

        total = 0
        for s in valid:
            if (prev_state & s) == 0 and \
                    (prev_state & (s << 1)) == 0 and \
                    (prev_state & (s >> 1)) == 0:
                total += dfs(row + 1, s, cnt + pc[s])
        return total

    print(dfs(0, 0, 0))

t = 1
# t = II()
for _ in range(t):
    solve()
