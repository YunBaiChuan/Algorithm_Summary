**思路：思维。主要是观察到题目所说的个位数递推关系式，这个我们其实可以枚举a1和a2的个位数，也就是10 \* 10的量级，接着再根据递推式验证一遍是否满足条件，然后从区间[l, r]中复原出对应的a1和a2即可，并且判断时要取复原数字的最小字典序作为答案**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

def solve():
    n, l1, r1, l2, r2 = LII()
    a = LII()

    m = n - 2
    ans1, ans2 = -1, -1

    for x in range(10):
        for y in range(10):
            p, q = x, y
            ok = True
            for i in range(m):
                cur = (p * q) % 10
                if cur != a[i]:
                    ok = False
                    break
                p, q = q, cur
            if not ok:
                continue

            v1 = l1 + ((x - l1) % 10 + 10) % 10
            if v1 > r1:
                continue
            v2 = l2 + ((y - l2) % 10 + 10) % 10
            if v2 > r2:
                continue

            if ans1 == -1 or (v1, v2) < (ans1, ans2):
                ans1, ans2 = v1, v2

    print(ans1, ans2)

# t = 1
t = II()
for _ in range(t):
    solve()
```
