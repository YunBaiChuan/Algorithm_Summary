**思路：dfn序 + 前后缀分解。对于计算去掉子树的某一部分值来说，用dfn序是很方便的，接着再利用前后缀分解完成本题**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if y > x else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, q = MII()
    a = [0] + LII()
    p = LII()

    g = [[] for _ in range(n + 1)]
    par = [0] * (n + 1)
    ch = [0] * (n + 1)
    for i in range(2, n + 1):
        fa = p[i - 2]
        par[i] = fa
        ch[fa] += 1
        g[fa].append(i)

    pre = [0] * (n + 1)
    lo = [0] * (n + 1)
    hi = [0] * (n + 1)
    pre[1] = a[1]
    order = []
    lp = []
    k = 0

    st = [(1, 0)]
    while st:
        u, f = st.pop()
        if f:
            hi[u] = k - 1
            order.append(u)
        else:
            lo[u] = k
            if ch[u] == 0:
                lp.append(pre[u])
                k += 1
            st.append((u, 1))
            for c in g[u]:
                pre[c] = pre[u] + a[c]
                st.append((c, 0))

    mx = [0] * (n + 1)
    for u in order:
        t = 0
        for c in g[u]:
            v = a[c] + mx[c]
            if v > t:
                t = v
        mx[u] = t

    L = len(lp)
    pf = [-1] * (L + 1)
    for i in range(L):
        pf[i + 1] = pf[i] if pf[i] > lp[i] else lp[i]
    sf = [-1] * (L + 1)
    for i in range(L - 1, -1, -1):
        sf[i] = sf[i + 1] if sf[i + 1] > lp[i] else lp[i]

    out = []
    for _ in range(q):
        x, y = MII()
        r = pre[y] + a[x] + mx[x]

        t = pf[lo[x]]
        if sf[hi[x] + 1] > t:
            t = sf[hi[x] + 1]
        if t > r:
            r = t

        fa = par[x]
        if ch[fa] == 1 and pre[fa] > r:
            r = pre[fa]

        out.append(str(r))

    print("\n".join(out))

t = 1
# t = II()
for _ in range(t):
    solve()
```
