## 第一题：爬楼梯 II https://leetcode.cn/problems/climbing-stairs-ii/description/?envType=daily-question&envId=2026-11-10

**思路：一维dp。根据题意，从后往前进行记忆化搜索，写出边界条件、递推式子，算结果即可**

**Python代码：**
```python
class Solution:
    def climbStairs(self, n: int, costs: List[int]) -> int:
        costs = [0] + costs
        @cache 
        def dfs(i):
            if i < 0:
                return inf 
            if i == 0:
                return 0 
            return min(dfs(i - 1) + 1, dfs(i - 2) + 4, dfs(i - 3) + 9) + costs[i]
        return dfs(n)
```

## 第二题：区间加乘与单点求值 https://www.nowcoder.com/practice/7a1de22fa7a1456f8ba519f21de31c84?channelPut=tracker2

**思路：线段树。这题乍一看很像是树状数组的区间修改、单点查询问题，不过这题里面有个操作是乘法、但差分对乘法来说是无效的（主要是边界除法有问题），因此只能用线段树来解决**

mod = 998244353

class SegTree:
    def __init__(self, n, a):
        self.n = n
        self.val = [0] * (4 * n)
        self.add = [0] * (4 * n)
        self.mul = [1] * (4 * n)
        self._build(1, 1, n, a)

    def _build(self, p, l, r, a):
        if l == r:
            self.val[p] = a[l] % mod
            return
        m = (l + r) // 2
        self._build(p * 2, l, m, a)
        self._build(p * 2 + 1, m + 1, r, a)

    def _push(self, p):
        for c in (p * 2, p * 2 + 1):
            self.val[c] = (self.val[c] * self.mul[p] + self.add[p]) % mod
            self.mul[c] = self.mul[c] * self.mul[p] % mod
            self.add[c] = (self.add[c] * self.mul[p] + self.add[p]) % mod
        self.mul[p] = 1
        self.add[p] = 0

    def range_add(self, L, R, x, p=1, l=1, r=None):
        if r is None:
            r = self.n
        if L <= l and r <= R:
            self.add[p] = (self.add[p] + x) % mod
            self.val[p] = (self.val[p] + x) % mod
            return
        self._push(p)
        m = (l + r) // 2
        if L <= m:
            self.range_add(L, R, x, p * 2, l, m)
        if R > m:
            self.range_add(L, R, x, p * 2 + 1, m + 1, r)

    def range_mul(self, L, R, x, p=1, l=1, r=None):
        if r is None:
            r = self.n
        if L <= l and r <= R:
            self.mul[p] = self.mul[p] * x % mod
            self.add[p] = self.add[p] * x % mod
            self.val[p] = self.val[p] * x % mod
            return
        self._push(p)
        m = (l + r) // 2
        if L <= m:
            self.range_mul(L, R, x, p * 2, l, m)
        if R > m:
            self.range_mul(L, R, x, p * 2 + 1, m + 1, r)

    def query(self, idx, p=1, l=1, r=None):
        if r is None:
            r = self.n
        if l == r:
            return self.val[p] % mod
        self._push(p)
        m = (l + r) // 2
        if idx <= m:
            return self.query(idx, p * 2, l, m)
        else:
            return self.query(idx, p * 2 + 1, m + 1, r)

def solve():
    n, q = MII()
    a = [0] + LII()

    tree = SegTree(n, a)
    out = []
    for _ in range(q):
        op, *b = LII()
        if op == 1:
            l, r, x = b
            tree.range_add(l, r, x % mod)
        elif op == 2:
            l, r, x = b
            tree.range_mul(l, r, x % mod)
        else:
            x = b[0]
            out.append(str(tree.query(x)))

    print('\n'.join(out))

t = 1
# t = II()
for _ in range(t):
    solve()
