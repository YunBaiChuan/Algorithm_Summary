**思路：思维。主要是算每个'#'位置的最大曼哈顿距离，也就是到某两个边界的最大距离，之后取这些曼哈顿距离中的最大值即为答案**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, m = MII()
    r = 0
    for i in range(n):
        s = I()
        a = fmax(i, n - 1 - i)
        p = s.find("#")
        while p != -1:
            b = fmax(p, m - 1 - p)
            r = fmax(r, a + b)
            p = s.find("#", p + 1)
    print(r)

t = 1
# t = II()
for _ in range(t):
    solve()
```
