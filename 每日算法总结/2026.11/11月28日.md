**思路：思维 + Kruskal。题目要算删边获得的最大价值等价于保留最小的价值，这个过程类似于最小生成树。具体来说，其价值由对应的2和5更小因子数决定。最后得到最小保留结果，用它去减总价值来得到结果**

**代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, m = MII()
    w = LII()

    def f(x, p):
        c = 0
        while x % p == 0:
            x //= p
            c += 1
        return c

    v2 = [f(w[i], 2) for i in range(n)]
    v5 = [f(w[i], 5) for i in range(n)]

    e = []
    tot = 0
    for _ in range(m):
        u, v = MII()
        u -= 1
        v -= 1
        a = v2[u] + v2[v]
        b = v5[u] + v5[v]
        c = a if a < b else b
        e.append((c, u, v))
        tot += c

    par = list(range(n))

    def fd(x):
        while par[x] != x:
            par[x] = par[par[x]]
            x = par[x]
        return x

    e.sort()
    s = 0
    for c, u, v in e:
        x = fd(u)
        y = fd(v)
        if x != y:
            par[x] = y
            s += c

    print(tot - s)

t = 1
# t = II()
for _ in range(t):
    solve()
```
