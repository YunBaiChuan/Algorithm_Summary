**思路：构造。首先预检查a数组是否存在必然无解的情况，简单来说初步合法的a数组需要具备两个条件：1.单调不增；2.前项可以被后项整除；检查完一遍之后就可以进行答案的构造了，从前往后，从小到大进行枚举构造，并且用数组来记录使用情况，最终构造出答案，并进行检查和输出即可**

import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    n = II()
    a = LII()

    for i in range(1, n):
        if a[i] > a[i - 1] or a[i - 1] % a[i] != 0:
            print(-1)
            return

    use = [0] * (n + 1)
    cur = [0] * (n + 1)
    p = [0] * n

    for i in range(n):
        g = a[i]
        q = a[i - 1] if i and a[i] < a[i - 1] else 0
        v = 0
        x = cur[g]
        while x <= n:
            x += g
            if x > n:
                break
            cur[g] = x
            if not use[x] and (q == 0 or x % q != 0):
                v = x
                break
        if v == 0:
            print(-1)
            return
        use[v] = 1
        p[i] = v

    print(*p)

t = 1
# t = II()
for _ in range(t):
    solve()
