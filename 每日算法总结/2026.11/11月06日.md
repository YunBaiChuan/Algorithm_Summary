## 第一题：小A的最短路 https://www.nowcoder.com/practice/c48b0cc8826c4770a0c746e62cc0ef90?channelPut=tracker2

**思路：思维 + lca。树上的最短路问题，首先要想到用lca进行解决，那算lca通常就是倍增st表预处理。之后就可以算一下三种可能的答案，然后取其中的最小值即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n = II()
    g = [[] for _ in range(n + 1)]
    for _ in range(n - 1):
        u, v = MII()
        g[u].append(v)
        g[v].append(u)
    u0, v0 = MII()

    LOG = (n).bit_length()
    up = [[0] * (n + 1) for _ in range(LOG)]
    depth = [0] * (n + 1)

    parent = [0] * (n + 1)
    parent[1] = 1
    order = [1]
    for x in order:
        for y in g[x]:
            if y != parent[x]:
                parent[y] = x
                depth[y] = depth[x] + 1
                up[0][y] = x
                order.append(y)

    for j in range(1, LOG):
        uj = up[j]
        uj1 = up[j - 1]
        for i in range(1, n + 1):
            uj[i] = uj1[uj1[i]]

    def lca(a, b):
        if depth[a] < depth[b]:
            a, b = b, a
        diff = depth[a] - depth[b]
        for j in range(LOG):
            if diff >> j & 1:
                a = up[j][a]
        if a == b:
            return a
        for j in range(LOG - 1, -1, -1):
            if up[j][a] != up[j][b]:
                a = up[j][a]
                b = up[j][b]
        return up[0][a]

    def dist(a, b):
        c = lca(a, b)
        return depth[a] + depth[b] - 2 * depth[c]

    out = []
    q = II()
    for _ in range(q):
        s, e = MII()
        # 情况1：不走免费边
        ans = dist(s, e)
        # 情况2：s -> u0 -> v0 -> e（免费边 u0-v0）
        ans = fmin(ans, dist(s, u0) + dist(v0, e))
        # 情况3：s -> v0 -> u0 -> e（免费边反向使用）
        ans = fmin(ans, dist(s, v0) + dist(u0, e))
        out.append(str(ans))

    print('\n'.join(out))

t = 1
# t = II()
for _ in range(t):
    solve()
```
