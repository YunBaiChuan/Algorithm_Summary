**思路：单调栈。很标准的单调栈板子题，这里只用到了right数组，即找右侧满足条件的元素索引**

**Python代码：**

import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    n = II()
    h = [0] + [II() for _ in range(n)]

    st = []
    right = [0] * (n + 1)
    for i in range(n, -1, -1):
        x = h[i]
        while st and h[st[-1]] <= x:
            st.pop()
        if st:
            right[i] = st[-1]
        st.append(i)

    print('\n'.join(map(str, right[1:])))

t = 1
# t = II()
for _ in range(t):
    solve()
