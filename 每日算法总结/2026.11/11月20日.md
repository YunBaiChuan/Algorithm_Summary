## 第一题：小红的矩阵修改 https://www.nowcoder.com/practice/e556fd4852954529aec85940801200a1?channelPut=tracker2

**思路：状压dp。按列进行状压dp，约束是行和列的字符不能重复，按列进行状压dp的原因在于n比较小，用n的话状态数更少**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, m = MII()
    s = [I() for _ in range(n)]

    mp = {'r': 0, 'e': 1, 'd': 2}
    a = [[mp[s[i][j]] for i in range(n)] for j in range(m)]

    sz = 3 ** n

    dig = [[0] * n for _ in range(sz)]
    ok = [False] * sz
    for st in range(sz):
        x = st
        good = True
        for i in range(n):
            d = x % 3
            dig[st][i] = d
            x //= 3
            if i and d == dig[st][i - 1]:
                good = False
        ok[st] = good

    f = [inf] * sz
    for st in range(sz):
        if not ok[st]:
            continue
        c = 0
        for i in range(n):
            if dig[st][i] != a[0][i]:
                c += 1
        f[st] = c

    for j in range(1, m):
        g = [inf] * sz
        base = [0] * sz
        for st in range(sz):
            if not ok[st]:
                continue
            c = 0
            for i in range(n):
                if dig[st][i] != a[j][i]:
                    c += 1
            base[st] = c

        for st in range(sz):
            if not ok[st]:
                continue
            best = inf
            for pr in range(sz):
                if f[pr] >= inf:
                    continue
                bad = False
                for i in range(n):
                    if dig[st][i] == dig[pr][i]:
                        bad = True
                        break
                if not bad:
                    val = f[pr] + base[st]
                    if val < best:
                        best = val
            g[st] = best
        f = g

    print(min(f))

t = 1
# t = II()
for _ in range(t):
    solve()
```
