**思路：Kosaraju。主要是利用Kosaraju找强连通分量，并且由于此题不限制经过点次数，因此并不是找哈密顿回路。找到每一个强连通分量之后，其中的最小权重就是答案，逐个累加即可；方案数就是各个强连通分量中，最小权重数量的累乘取模。最终，输出答案即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

mod = 10 ** 9 + 7

def solve():
    n = II()
    w = [0] + LII()
    m = II()

    g = [[] for _ in range(n + 1)]
    rg = [[] for _ in range(n + 1)]
    for _ in range(m):
        x, y = MII()
        g[x].append(y)
        rg[y].append(x)

    vis = [False] * (n + 1)
    order = []
    for s in range(1, n + 1):
        if vis[s]:
            continue
        st = [(s, 0)]
        while st:
            u, phase = st.pop()
            if phase == 0:
                if vis[u]:
                    continue
                vis[u] = True
                st.append((u, 1))
                for v in g[u]:
                    if not vis[v]:
                        st.append((v, 0))
            else:
                order.append(u)

    comp = [0] * (n + 1)
    cid = 0
    for u in reversed(order):
        if comp[u]:
            continue
        cid += 1
        comp[u] = cid
        st = [u]
        while st:
            x = st.pop()
            for v in rg[x]:
                if not comp[v]:
                    comp[v] = cid
                    st.append(v)

    mn = [inf] * (cid + 1)
    cnt = [0] * (cid + 1)
    for i in range(1, n + 1):
        c = comp[i]
        if w[i] < mn[c]:
            mn[c] = w[i]
            cnt[c] = 1
        elif w[i] == mn[c]:
            cnt[c] += 1

    total_cost = 0
    ways = 1
    for c in range(1, cid + 1):
        total_cost += mn[c]
        ways = ways * cnt[c] % mod

    print(total_cost, ways)

t = 1
# t = II()
for _ in range(t):
    solve()
```
