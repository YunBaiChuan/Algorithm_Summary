import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, m = MII()
    a = [LII() for _ in range(n)]
    a.sort(key=lambda x: x[2])

    groups = []
    i = 0
    while i < n:
        group = []
        st = i
        while i < n and a[i][2] == a[st][2]:
            group.append((a[i][0], a[i][1]))
            i += 1
        groups.append(group)

    team = len(groups)
    dp = [[0] * (m + 1) for _ in range(team + 1)]

    for i in range(team):
        for j in range(m + 1):
            dp[i + 1][j] = dp[i][j]
            for v, w in groups[i]:
                if j >= v:
                    dp[i + 1][j] = fmax(dp[i + 1][j], dp[i][j - v] + w)

    print(dp[-1][-1])

t = 1
# t = II()
for _ in range(t):
    solve()
