def solve():
    n, m = MII()
    g = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = MII()
        g[u].append(v)
        g[v].append(u)

    size = [0] * (n + 1)
    sub = [0] * (n + 1)
    parent = [0] * (n + 1)

    order = []
    stack = [1]
    parent[1] = 0
    while stack:
        u = stack.pop()
        order.append(u)
        for v in g[u]:
            if v != parent[u]:
                parent[v] = u
                stack.append(v)

    for u in reversed(order):
        size[u] = 1
        sub[u] = 0
        for v in g[u]:
            if v != parent[u]:
                size[u] += size[v]
                sub[u] += sub[v] + size[v]

    dp = [0] * (n + 1)
    dp[1] = sub[1]
    for u in order:
        for v in g[u]:
            if v != parent[u]:
                dp[v] = dp[u] + n - 2 * size[v]

    mn = min(dp[1:])
    ans = []
    for i in range(1, n + 1):
        if dp[i] == mn:
            ans.append(i)
    print(*ans)

t = 1
# t = II()
for _ in range(t):
    solve()
