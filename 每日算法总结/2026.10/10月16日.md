**思路：思维 + 分组循环 + 贪心。本题要算的是非0偶数的最长子串长度，因此我们可以先用分组循环预处理出所有相同串的长度，然后再贪心遍历一遍，算最大非0偶数子串的长度，最后输出即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    s = I()
    tmp = []
    i = 0
    n = len(s)
    while i < n:
        st = i
        i += 1
        while i < n and s[i] == s[i - 1]:
            i += 1
        tmp.append(i - st)

    ans = res = 0
    for i in range(len(tmp)):
        if tmp[i] & 1:
            ans = fmax(ans, res)
            res = 0
            continue
        res += tmp[i]

    print(fmax(ans, res))

t = 1
# t = II()
for _ in range(t):
    solve()
```
