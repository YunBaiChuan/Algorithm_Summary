**思路：不定长滑动窗口。两个维度的不定长滑动窗口，越短越合法问题。在原有的基础上，再添加一个维度，然后修改一下while条件即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, m, u = LII()
    a = LII()
    b = LII()

    ans = l = 0
    s1 = s2 = 0
    for r in range(n):
        s1 += a[r]
        s2 += b[r]
        while s1 > m or s2 > u:
            s1 -= a[l]
            s2 -= b[l]
            l += 1
        ans = fmax(ans, r - l + 1)
    print(ans)

t = 1
# t = II()
for _ in range(t):
    solve()
```
