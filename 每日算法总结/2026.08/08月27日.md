**Cpp代码：**
```cpp
#include <iostream>
using namespace std;

using ll = long long;
const ll inf = 1e18;

void solve() {
    int a;
    cin >> a;
    if (a == 1) {
        cout << "2 3" << '\n';
    } else if (a == 1e9) {
        cout << "536870912 463129088" << '\n';
    } else {
        cout << (a ^ 1) << ' ' << 1 << '\n';
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    cin >> T;
    while (T--) {
        solve();
    }
}
```
