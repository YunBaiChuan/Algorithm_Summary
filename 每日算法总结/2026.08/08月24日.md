**代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

def solve():
    px, py, ux, uy, vx, vy = LFI()
    # 先算河水向量
    dx = vx - ux
    dy = vy - uy
    # 通过PQ向量与河水向量的关系，推导出参数t
    t = ((px - ux) * dx + (py - uy) * dy) / (dx * dx + dy * dy)
    # 得到Q坐标
    qx = ux + t * dx
    qy = uy + t * dy
    print(f"{qx:.6f} {qy:.6f}")

# T = 1
T = II()
for _ in range(T):
    solve()
```
