import sys
from collections import deque

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    n = II()
    g = [[] for _ in range(n + 1)]

    for i in range(n):
        u, v = MII()
        g[u].append((v, i))
        g[v].append((u, i))

    d = [len(g[i]) for i in range(n + 1)]
    q = deque(i for i in range(1, n + 1) if d[i] == 1)
    c = [1] * (n + 1)

    while q:
        u = q.popleft()
        c[u] = 0
        for v, _ in g[u]:
            d[v] -= 1
            if d[v] == 1:
                q.append(v)

    r = [0] * (n + 1)
    p = [0] * (n + 1)
    e = [-1] * (n + 1)
    h = [0] * (n + 1)

    q = deque()
    for i in range(1, n + 1):
        if c[i]:
            r[i] = i
            q.append(i)

    while q:
        u = q.popleft()
        for v, z in g[u]:
            if r[v] or c[v]:
                continue
            r[v] = r[u]
            p[v] = u
            e[v] = z
            h[v] = h[u] + 1
            q.append(v)

    a = [0] * n
    if r[1] == r[n]:
        x, y = 1, n
        while h[x] > h[y]:
            a[e[x]] = -1
            x = p[x]
        while h[y] > h[x]:
            a[e[y]] = -1
            y = p[y]
        while x != y:
            a[e[x]] = -1
            a[e[y]] = -1
            x = p[x]
            y = p[y]
        z = h[1] + h[n] - 2 * h[x]
        for i in range(n):
            if a[i] != -1:
                a[i] = z
    else:
        x = r[1]
        y = r[n]

        v = []
        f = []
        u = x
        w = 0
        while True:
            v.append(u)
            for t, z in g[u]:
                if c[t] and t != w:
                    f.append(z)
                    w, u = u, t
                    break
            if u == x:
                break

        k = len(v)
        j = v.index(y)
        z = h[1] + h[n]

        for i in range(n):
            a[i] = z + min(j, k - j)

        for i in range(j):
            a[f[i]] = z + k - j
        for i in range(j, k):
            a[f[i]] = z + j

        x = 1
        while not c[x]:
            a[e[x]] = -1
            x = p[x]

        x = n
        while not c[x]:
            a[e[x]] = -1
            x = p[x]

    print("\n".join(map(str, a)))


t = 1
# t = II()
for _ in range(t):
    solve()
