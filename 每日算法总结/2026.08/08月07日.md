**思路：枚举。枚举四种翻牌顺序，然后再模拟两回合，判断能否胜利即可。直接用$O(1)$公式判断比较麻烦，主要是一胜一平的情况比较难处理，建议直接枚举**

import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    a1, a2, b1, b2 = LII()

    cases = [
        (a1, b1, a2, b2),
        (a1, b2, a2, b1),
        (a2, b1, a1, b2),
        (a2, b2, a1, b1)
    ]

    ans = 0
    for x1, y1, x2, y2 in cases:
        aw = 0
        bw = 0

        if x1 > y1:
            aw += 1
        elif x1 < y1:
            bw += 1

        if x2 > y2:
            aw += 1
        elif x2 < y2:
            bw += 1

        if aw > bw:
            ans += 1

    print(ans)

# t = 1
t = II()
for _ in range(t):
    solve()
