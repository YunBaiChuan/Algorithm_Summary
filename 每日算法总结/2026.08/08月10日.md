## 第一题：石子游戏 IV https://leetcode.cn/problems/stone-game-iv/description/?envType=daily-question&envId=2026-08-10

**思路：博弈dp。这题的状态转移更容易想一些，就是枚举拿多少$x * x$石子，然后当前人要获胜的话，下一个人$dfs(i - x * x)$状态就必须输，就是比较经典的零和博弈**

**代码：**
```cpp
class Solution {
public:
    bool winnerSquareGame(int n) {
        vector<int8_t> memo(n + 1, -1); // 初始化n + 1空间，且为-1，不和0状态撞了。并且用int8_t省空间

        auto dfs = [&](this auto&& dfs, int i) -> bool { // 要显式指定返回值类型，编译器不会推导这个
            if (i == 0) {
                return false;
            }

            if (memo[i] != -1) {
                return memo[i];
            }

            for (int x = 1; x * x <= i; x++) {
                if (!dfs(i - x * x)) {
                    return memo[i] = true; // 简洁写法，表示先赋值再返回这个变量
                }
            }
            return memo[i] = false;
        };

        return dfs(n);
    }
};
```

## 第二题：小红的数组操作 https://www.nowcoder.com/practice/1ef66ea2e72e402680b4a443a8d02fd6?channelPut=tracker2

**思路：前后缀分解 + 滑动窗口 + 思维。首先预处理出前后缀染色数组，并且计算出其染色的最小代价；接着用滑动窗口找中间不重复的非染色区域，然后算出其左右两遍染色的代价；最终遍历一遍得到答案**

**代码：**
```cpp
#include <iostream> // cpp标准io库，主要是cin cout等
#include <algorithm> // cpp算法库，主要是sort max min等
#include <vector> // cpp的vector库，引入vector
#include <set> // cpp的set库，引入set
using namespace std;

using ll = long long;
const ll inf = 1e18;

void solve() {
    int n;
    cin >> n;
    
    vector<int> a(n);
    vector<ll> pre(n + 1);
    vector<ll> suf(n + 1);

    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    for (int i = 0; i < n; i++) {
        pre[i + 1] = (ll)a[i] * (i + 1);
        suf[i] = (ll)a[i] * (n - i);
    }
    for (int i = n - 1; i > 0; i--) {
        pre[i] = min(pre[i], pre[i + 1]);
    }
    for (int i = 1; i < n; i++) {
        suf[i] = min(suf[i], suf[i - 1]);
    }

    set<int> st;
    ll res = inf;
    int l = 0;
    for (int i = 0; i < n; i++) {
        while (st.count(a[i])) {
            st.erase(a[l]);
            l++;
        }
        st.insert(a[i]);
        res = min(res, pre[l] + suf[i + 1]);
    }

    cout << res << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) {
        solve();
    }
    return 0;
}
```
