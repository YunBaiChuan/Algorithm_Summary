## 第一题：找出最大的几近缺失整数 https://leetcode.cn/problems/find-the-largest-almost-missing-integer/description/?envType=daily-question&envId=2026-08-18

**思路：思维 + 分类讨论。对于k = n的情况来说，其结果就是整个数组的最大值；对于k = 1的情况来说，答案为个数是1元素中的最大值，这里可以用hash表统计，会比有序表更快；对于1 < k < n的情况来说，只有nums[0]和nums[-1]有可能成为答案，并且对于nums[0]来说，要使其成为答案候选，nums[1:]之后不能有和他相同的元素；nums[-1]同理，就是nums[:-1]不能有和他相同的元素。判断好之后，其中的较大值为答案**

**代码：**
```cpp
class Solution {
public:
    int largestInteger(vector<int>& nums, int k) {
        int n = nums.size();
        if (k == n) {
            return *max_element(nums.begin(), nums.end());
        }
        if (k == 1) {
            // 这里的unordered_map相当于是python的字典，他是无序的hash表，map的话是有序表，时间复杂度会更高点
            unordered_map<int, int> cnt;
            for (int x: nums) {
                cnt[x]++;
            }
            int ans = -1;
            for (auto& [x, c]: cnt) {
                if (c == 1) {
                    ans = max(ans, x);
                }
            }
            return ans;
        }

        auto f = [&](int begin, int end, int x) {
            // find函数，如果没找的就返回查找范围的end迭代器
            if (find(nums.begin() + begin, nums.begin() + end, x) != nums.begin() + end) {
                return -1;
            }
            return x;
        };

        // nums[0]和nums[n - 1]只有出现一次才能参与最终答案的计算
        return max(f(1, n, nums[0]), f(0, n - 1, nums[n - 1]));
    }
};
```

## 第二题：茜茜的计算器 https://www.nowcoder.com/practice/d2d4711a6f20490f814e5fc6f2e64aac?channelPut=tracker2

**思路：思维 + 组合数学。首先考虑上下对称，那就是0、1、3、8这四个数在n个位置上任意放，结果就是pow(4, n)；再考虑左右对称，那就是0、(2、3)、8这四个数在n个位置上对称放，但此时就要考虑n的奇偶性了。对于n是偶数来说，我们考虑放一半位置，另一半是一一对应的。那么方案数就是pow(4, n / 2)，但要注意我们多放了一次0、8，因此要减去pow(2, n / 2)；同理，对于n是奇数来说，就是中间位置可以放0、8，其实就是偶数时的2倍。最终，输出答案即可**

**代码：**
```cpp
#include <iostream>
using namespace std;

using ll = long long;
const ll inf = 1e18;
const ll mod = 1e9 + 7;

ll qpow(ll x, ll k) {
    ll res = 1;
    while (k) {
        if (k & 1) res = res * x % mod;
        x = x * x % mod;
        k >>= 1;
    }
    return res % mod;
}

void solve() {
    int n;
    cin >> n;
    if (n & 1) {
        cout << (qpow(4, n) + (qpow(4, n / 2) - qpow(2, n / 2) + mod) * 2 ) % mod << '\n';
        return;
    }
    cout << (qpow(4, n) + qpow(4, n / 2) - qpow(2, n / 2) + mod) % mod << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) {
        solve();
    }
    return 0;
}
```
