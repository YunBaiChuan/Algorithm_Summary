**Cpp代码：**
```cpp
#include <iostream>
#include <unordered_map>
#include <vector>
using namespace std;

using ll = long long;
const ll inf = 1e18;

void solve() {
    int n;
    cin >> n;
    string s;
    cin >> s;

    unordered_map<char, char> d1;
    for (int i = 0; i < 13; i++) {
        char key = 'a' + 25 - i;
        char val = 'a' + i;
        d1[key] = val;
    }

    unordered_map<char, char> d2;
    for (int i = 0; i < 13; i++) {
        char key = 'A' + i;
        char val = 'A' + 25 - i;
        d1[key] = val;
    }

    vector<char> st;
    for (auto x: s) {
        if (st.empty()) {
            st.push_back(x);
            continue;
        }

        auto it1 = d1.find(x);
        if (it1 != d1.end() && st.back() == d1[x]) {
            st.pop_back();
            continue;
        }

        auto it2 = d2.find(x);
        if (it2 != d2.end() && st.back() == d2[x]) {
            st.pop_back();
            continue;
        }

        st.push_back(x);
    }

    cout << st.size() << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) {
        solve();
    }
}
```
