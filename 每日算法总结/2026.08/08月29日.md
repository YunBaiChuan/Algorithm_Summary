**思路；模拟。就先写个m次的训话，然后判断当前的3种操作结果中，哪个更小就选哪个作为下一轮输入。当n <= 3时，操作2的-1就更占优势了，直接输出答案即可**

**Cpp代码：**
```cpp
#include <iostream>
#include <cmath>
using namespace std;

using ll = long long;
const ll inf = 1e18;

void solve() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        if (n <= 3) {
            cout << n - (m - i) << '\n';
            return;
        }
        int p1, p2, p3;
        p1 = ceil(sqrt(n));
        p2 = n - 1;
        p3 = ceil(n / 2.0);
        n = min(min(p1, p2), p3);
    }
    cout << n << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    cin >> T;
    while (T--) {
        solve();
    }
}
```
