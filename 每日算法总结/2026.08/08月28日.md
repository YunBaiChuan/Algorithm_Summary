## 第一题：小苯的能量项链 https://www.nowcoder.com/practice/375d5b2068dd402fa53c73a2418927c0?channelPut=tracker2

**思路：思维 + 前后缀分解。这题有个小坑在于我们不是必须要用完所有操作，而是可以只用其中的一小部分。所以说就可以预处理出一个后缀和数组，然后枚举操作数，不断算最大值即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, k = MII()
    a = LII()
    if n < 3:
        print(sum(a))
        return

    op = fmin(n - 2, k)

    suf = [0] * n
    suf[-1] = a[-1]
    for i in range(n - 2, -1, -1):
        suf[i] = fmax(a[i], suf[i + 1])

    ans = 0
    for i in range(op + 1):
        r_del = op - i
        r_start = n - 1 - r_del
        ans = fmax(ans, a[i] + suf[r_start])

    print(ans)

# t = 1
t = II()
for _ in range(t):
    solve()
```

**Cpp代码：**
```cpp
#include <iostream>
#include <vector>
#include <numeric>
using namespace std;

using ll = long long;
const ll inf = 1e18;

void solve() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (auto& x: a) {
        cin >> x;
    }

    if (n < 3) {
        cout << accumulate(a.begin(), a.end(), 0) << '\n';
        return;
    }

    vector<int> suf(n);
    suf[n - 1] = a[n - 1];
    for (int i = n - 2; i >= 0; i--) {
        suf[i] = max(a[i], suf[i + 1]);
    }

    int op = min(n - 2, k);
    int ans = 0;
    for (int i = 0; i < op + 1; i++) {
        ans = max(ans, a[i] + suf[n - 1 - (op - i)]);
    }

    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    cin >> T;
    while (T--) {
        solve();
    }
}
```
