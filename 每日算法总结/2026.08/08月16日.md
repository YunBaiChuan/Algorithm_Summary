## 第一题：石子游戏 IX https://leetcode.cn/problems/stone-game-ix/description/?envType=daily-question&envId=2026-08-16

**思路：思维。把玩样例可知，状态%3之后其实只有0，1，2，我们想要获胜的话，每次不能为0。因此就可以从两个方向进行思考：1.如果开始选1，那么博弈序列应该为1 1 2 1 2...；如果开始选2，那么博弈序列应该为2 2 1 2 1...。因此，我们就分别算一下这两种博弈序列能够进行多少论，如果是奇数轮，说明Alice能够获胜，否则为Bob。最终，这两种情况有一个成立即可**

**代码：**
```cpp
class Solution {
private:
    bool check(int n, int cnt[3]) {
        if (cnt[1] == 0) {
            return false;
        }
        cnt[1]--;
        int round = 1 + min(cnt[1], cnt[2]) * 2 + cnt[0];
        if (cnt[1] > cnt[2]) {
            round++;
        }
        return round < n && round % 2 != 0;
    }
public:
    bool stoneGameIX(vector<int>& stones) {
        int cnt[3]{}; // 表示创建长度为3的全0初始化数组
        for (int x: stones) {
            cnt[x % 3]++;
        }

        int n = stones.size();
        int cnt2[3] = {cnt[0], cnt[2], cnt[1]};
        return check(n, cnt) || check(n, cnt2);
    }
};
```

## 第二题：网络延迟时间 https://leetcode.cn/problems/network-delay-time/description/

**思路：Dijkstra。标准的dijkstra板子题，这个cpp版本需要记住，其实和python版本的本质相同，就是一些语法不同。需要注意以下东西：1.整个算法中的变量类型声明最好一致，比如用int就全是int，不要一个int一个long，会有截断溢出，导致错误；2.vector的*max_element找max元素方法可以调找的区域，比如这里用的就是dis.begin() + 1**

**代码：**
```cpp
class Solution {
public:
    int networkDelayTime(vector<vector<int>>& times, int n, int k) {
        int inf = 1e9; // 后续都用的int，所以说这里用int比较好，用long long容易发生截断溢出
        vector<vector<pair<int, int>>> g(n + 1);
        for (auto& t: times) {
            g[t[0]].emplace_back(t[1], t[2]);
        }

        vector<int> dis(n + 1, inf);
        dis[k] = 0;
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> h; // 小根堆标准定义，中间不能跳过vector参数直接写greater，会编译错误。cpp默认是大根堆
        h.emplace(0, k);
        while (!h.empty()) {
            auto [d, x] = h.top();
            h.pop();
            if (d > dis[x]) {
                continue;
            }
            for (auto& [y, w]: g[x]) {
                int new_dis = d + w; 
                if (new_dis < dis[y]) {
                    dis[y] = new_dis;
                    h.emplace(new_dis, y);
                }
            }
        }

        int mx = *max_element(dis.begin() + 1, dis.end());
        return mx < inf ? mx : -1;
    }
};
```

## 第三题：小红送外卖 https://www.nowcoder.com/practice/2850d7c941f6494e82ba74bc899eb512?channelPut=tracker2

**思路：Dijkstra。标准的cpp版本dijksta板子题，和常规相比就多了个无向图，对于无向图来说，来回的最短距离一样，那就*2即可，其他写法一致**

**代码：**
```cpp
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

using ll = long long;
const ll inf = 1e18;

void solve() {
    ll n, m, q;
    cin >> n >> m >> q;
    ll u, v, w;
    vector<vector<pair<ll, ll>>> g(n + 1);
    for (int i = 0; i < m; i++) {
        cin >> u >> v >> w;
        g[u].emplace_back(v, w);
        g[v].emplace_back(u, w);
    }
    vector<ll> a(q);
    for (int i = 0; i < q; i++) {
        cin >> a[i];
    }

    vector<ll> dis(n + 1, inf); // 记得初始化
    dis[1] = 0;
    priority_queue<pair<ll, ll>, vector<pair<ll, ll>>, greater<>> h; // 小根堆标准定义，中间不能跳过vector参数直接写greater，会编译错误。cpp默认是大根堆
    h.emplace(0, 1);
    while (!h.empty()) {
        auto [d, x] = h.top();
        h.pop();
        if (d > dis[x]) {
            continue;
        }
        for (auto& [y, w]: g[x]) { // 用auto&引用，防止用auto拷贝数据，造成空间浪费
            ll new_dis = d + w;
            if (new_dis < dis[y]) {
                dis[y] = new_dis;
                h.emplace(new_dis, y);
            }
        }
    }

    ll ans = 0;
    for (int i = 0; i < q; i++) {
        ans += 2 * dis[a[i]];
    }
    cout << ans << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) {
        solve();
    }
    return 0;
}
```
