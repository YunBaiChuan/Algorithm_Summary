## 第一题：石子游戏 V https://leetcode.cn/problems/stone-game-v/description/?envType=daily-question&envId=2026-08-17

**思路：博弈dp + 前缀和优化。有点像区间dp的枚举方式，然后状态转移是采用题目描述的博弈方式，并且涉及到区间和，需要采用前缀和进行优化**

**代码：**
```cpp
class Solution {
public:
    int stoneGameV(vector<int>& stoneValue) {
        int n = stoneValue.size();
        vector<int> sum(n + 1); // stoneValue 的前缀和
        partial_sum(stoneValue.begin(), stoneValue.end(), sum.begin() + 1);
        vector<vector<int>> memo(n, vector<int>(n + 1, -1));

        // 返回 Alice 在子数组 [i,j) 中获得的最大分数
        auto dfs = [&](this auto&& dfs, int i, int j) -> int {
            if (j - i == 1) { // 只剩下一块石子，游戏结束
                return 0;
            }

            int& res = memo[i][j]; // 注意这里是引用
            if (res != -1) { // 之前计算过
                return res;
            }

            // 把子数组 [i,j) 分成 [i,k) 和 [k,j)
            for (int k = i + 1; k < j; k++) {
                int sum_l = sum[k] - sum[i];
                int sum_r = sum[j] - sum[k];
                int score;
                if (sum_l < sum_r) { // Bob 丢弃 [k,j)，剩下 [i,k)
                    score = dfs(i, k) + sum_l;
                } else if (sum_l > sum_r) { // Bob 丢弃 [i,k)，剩下 [k,j)
                    score = dfs(k, j) + sum_r;
                } else { // sum_l = sum_r，由 Alice 决定丢弃哪边
                    score = max(dfs(i, k), dfs(k, j)) + sum_l;
                }
                res = max(res, score);
            }

            return res;
        };

        return dfs(0, n);
    }
};
```

## 第二题：睡前床边看 Live https://www.nowcoder.com/practice/8eef143409ae4ed9963f68bbe4b8e2a7?channelPut=tracker2

**思路：思维。由题意可知，每个人可以看到的数要么是众数，要么是众数-1。因此，如果人数差值> 1说明无解；当所有人的颜色相同时，要么整个为一组mx = n - 1成立，要么分多组mx * 2 <= n成立；然后就是两个数的情况，此时最小值个数应该等于mx才成立**

**代码：**
```cpp
#include <iostream>
#include <vector>
using namespace std;

using ll = long long;
const ll inf = 1e18;

int n;
int mx;
int mn;
int cntMn;

bool check() {
    if (mx - mn > 1) {
        return false;
    }
    if (mx == mn) {
        return mx == n - 1 || mx * 2 <= n;
    }
    return cntMn == mx;
}

void solve() {
    cin >> n;
    mn = n;
    mx = 0;
    cntMn = 0;
    for (int i = 0; i < n; i++) {
        int x;
        cin >> x;
        mx = max(mx, x);
        if (x < mn) {
            mn = x;
            cntMn = 1;
        } else if (x == mn) {
            cntMn++;
        }
    }
    cout << (check() ? "Other\n" : "Lie\n");
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    cin >> T;
    while (T--) {
        solve();
    }
    return 0;
}
```
