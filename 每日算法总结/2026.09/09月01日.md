## 第一题：小红的排列构造 https://www.nowcoder.com/practice/f83be390a5d4495f9823ee386be73172?channelPut=tracker2

**思路：构造。首先有个特判，由于是两个排列，所以说不可能构造出3个及以上的相同元素；然后前向模拟构造一遍p，后向模拟构造一遍q，然后检查下构造出来的p，q是否满足条件即可**

**Python代码：**
```python
import sys
from collections import Counter

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

def solve():
    n = II()
    a = [0] + LII()

    cnt = Counter(a)
    if max(cnt.values()) > 2:
        print(-1)
        return

    vis = [False] * (n + 1)
    p = [-1] * (n + 1)
    for i in range(1, n + 1):
        if not vis[a[i]]:
            p[i] = a[i]
            vis[a[i]] = True
    i = j = 1
    while i < n + 1 and j < n + 1:
        if vis[i]:
            i += 1
            continue
        if p[j] != -1:
            j += 1
            continue
        p[j] = i
        vis[i] = True

    vis = [False] * (n + 1)
    q = [-1] * (n + 1)
    for i in range(n, 0, -1):
        if not vis[a[i]]:
            q[i] = a[i]
            vis[a[i]] = True
    i = j = 1
    while i < n + 1 and j < n + 1:
        if vis[i]:
            i += 1
            continue
        if q[j] != -1:
            j += 1
            continue
        q[j] = i
        vis[i] = True

    for i in range(1, n + 1):
        if a[i] != p[i] and a[i] != q[i]:
            print(-1)
            return

    print(*p[1:])
    print(*q[1:])

t = 1
# t = II()
for _ in range(t):
    solve()
```
