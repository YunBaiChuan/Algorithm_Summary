**思路：网格图dp。这题和简单的网格图dp不同，它首先多了个右下可移动，并且三个方向在没有阻挡的情况下，可以移动任意步数**

**Python代码：**
```python
import sys
from collections import defaultdict

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else x
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, m = MII()
    M = [list(I()) for _ in range(n)]

    dp = [[inf] * m for _ in range(n)]
    row = [inf] * n
    line = [inf] * m
    dic = defaultdict(lambda: inf)

    dp[0][0] = row[0] = line[0] = dic[0] = 0

    for i in range(n):
        for j in range(m):
            if i == 0 and j == 0:
                continue
            if M[i][j] == '*':
                row[i] = inf
                line[j] = inf
                dic[i - j] = inf
            else:
                dp[i][j] = min(row[i], line[j], dic[i - j]) + 1
                row[i] = min(row[i], dp[i][j])
                line[j] = min(line[j], dp[i][j])
                dic[i - j] = min(dic[i - j], dp[i][j])

    ans = dp[n - 1][m - 1]
    print(ans if ans < inf else -1)

t = 1
# t = II()
for _ in range(t):
    solve()
```
