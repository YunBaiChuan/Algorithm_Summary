import sys

import math
inf = 10 ** 18

    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y

def solve():
    n, m = MII()

    g = [[] for _ in range(n + 1)]
    rg = [[] for _ in range(n + 1)]

    for _ in range(m):
        a, b = MII()
        g[a].append(b)
        rg[b].append(a)

    vis = [0] * (n + 1)
    order = []

    for i in range(1, n + 1):
        if vis[i]:
            continue

        st = [(i, 0)]
        while st:
            u, ph = st.pop()

            if ph == 0:
                if vis[u]:
                    continue
                vis[u] = 1
                st.append((u, 1))
                for v in g[u]:
                    if not vis[v]:
                        st.append((v, 0))
            else:
                order.append(u)

    comp = [0] * (n + 1)
    sz = []
    cid = 0

    for u in reversed(order):
        if comp[u]:
            continue

        cid += 1
        comp[u] = cid
        cnt = 1
        st = [u]

        while st:
            x = st.pop()
            for v in rg[x]:
                if not comp[v]:
                    comp[v] = cid
                    cnt += 1
                    st.append(v)

        sz.append(cnt)

    outdeg = [0] * (cid + 1)
    for u in range(1, n + 1):
        cu = comp[u]
        for v in g[u]:
            cv = comp[v]
            if cu != cv:
                outdeg[cu] += 1

    sink = 0
    for i in range(1, cid + 1):
        if outdeg[i] == 0:
            if sink == 0:
                sink = i
            else:
                print(0)
                return

    if sink == 0:
        print(0)
        return

    print(sz[sink - 1])

t = 1
# t = II()
for _ in range(t):
    solve()
