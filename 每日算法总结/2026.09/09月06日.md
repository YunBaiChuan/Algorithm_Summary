class FHQTreap:
    def __init__(self, maxn):
        self.a = [0] * maxn
        self.b = [0] * maxn
        self.c = [0] * maxn
        self.l = [0] * maxn
        self.r = [0] * maxn
        self.p = [0] * maxn
        self.n = 0
        self.seed = 123456789
        self.root = 0
        self.cnt = {}
        self.ans = 1

        self.root = self.new(1, 100000, 0)

    def new(self, x, y, v):
        self.n += 1
        self.seed = (self.seed * 1103515245 + 12345) & 2147483647
        self.a[self.n] = x
        self.b[self.n] = y
        self.c[self.n] = v
        self.p[self.n] = self.seed
        self.l[self.n] = 0
        self.r[self.n] = 0
        return self.n

    def merge(self, x, y):
        if x == 0:
            return y
        if y == 0:
            return x
        if self.p[x] > self.p[y]:
            self.r[x] = self.merge(self.r[x], y)
            return x
        self.l[y] = self.merge(x, self.l[y])
        return y

    def split(self, x, k):
        if x == 0:
            return 0, 0
        if self.b[x] < k:
            u, v = self.split(self.r[x], k)
            self.r[x] = u
            return x, v
        if self.a[x] >= k:
            u, v = self.split(self.l[x], k)
            self.l[x] = v
            return u, x
        y = self.b[x]
        v = self.c[x]
        u = self.l[x]
        w = self.r[x]
        self.l[x] = 0
        self.r[x] = 0
        self.b[x] = k - 1
        x = self.merge(u, x)
        t = self.new(k, y, v)
        t = self.merge(t, w)
        return x, t

    def add_cnt(self, x, v):
        self.cnt[x] = self.cnt.get(x, 0) + v

    def remove_cnt(self, x, v):
        self.cnt[x] -= v
        if self.cnt[x] == 0:
            del self.cnt[x]
            self.ans -= 1

    def traverse(self, x):
        if x == 0:
            return
        stack = [x]
        while stack:
            i = stack.pop()
            if i == 0:
                continue
            stack.append(self.l[i])
            stack.append(self.r[i])
            val = self.c[i]
            if val != 0:
                self.cnt[val] -= self.b[i] - self.a[i] + 1
                if self.cnt[val] == 0:
                    del self.cnt[val]
                    self.ans -= 1

    def assign(self, x, y, v):
        u, w = self.split(self.root, x)
        w, t = self.split(w, y + 1)
        self.traverse(w)
        if v != 0:
            if self.cnt.get(v, 0) == 0:
                self.ans += 1
            self.cnt[v] = self.cnt.get(v, 0) + y - x + 1
        self.root = self.merge(self.merge(u, self.new(x, y, v)), t)

    def query(self):
        return self.ans

def solve():
    q = II()
    treap = FHQTreap(300005)
    out = []
    for _ in range(q):
        l, r, x = LII()
        treap.assign(l, r, x)
        out.append(str(treap.query()))
    sys.stdout.write("\n".join(out))

t = 1

for _ in range(t):
    solve()
