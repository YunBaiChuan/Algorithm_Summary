**思路：构造题。先找第一个非零元素和最后一个非零元素，然后进行对应的左侧零填充以及右侧零填充；中间部分就从左往右填充，如果为零就以左侧非零元素为准进行填充；之后判断一下两个非零元素是否相同，相同的话可以把$a[0] += 1$；最终整体过一遍，判断是否满足条件即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    n = II()
    a = LII()

    l = 0
    while l < n and a[l] == 0:
        l += 1

    if l == n:
        a = [1] * n
        a[0] = 2
        print(*a)
        return

    r = n - 1
    while r >= 0 and a[r] == 0:
        r -= 1

    for i in range(l):
        a[i] = a[l]

    for i in range(r + 1, n):
        a[i] = a[r]

    if a[l] == a[r]:
        if l == 0:
            if r == n - 1:
                print(-1)
                return
            a[-1] = a[r] + 1
        else:
            a[0] = a[l] + 1

    for i in range(l + 1, r):
        if a[i] == 0:
            a[i] = a[i - 1]

    diff = 0
    for i in range(1, n):
        diff += abs(a[i] - a[i - 1])
        if diff > 1:
            print(-1)
            return

    if diff == 0:
        print(-1)
        return
    print(*a)

t = 1
# t = II()
for _ in range(t):
    solve()
```
