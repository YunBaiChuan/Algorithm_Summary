**思路：质因数分解 + 二分。如果直接暴力枚举n的话肯定会超时，考虑质因数分解。我们要使得n!为p的倍数的话，那么n!至少要包含p的所有质因子个数。比如p有3个2，2个3的话，我们的n!至少也要有3个2，2个3。所以说我们进行质因数分解，然后用二分查找满足条件的n!即可**

def prime_factors(x):
    factors = []
    d = 2
    while d * d <= x:
        if x % d == 0:
            cnt = 0
            while x % d == 0:
                x //= d
                cnt += 1
            factors.append((d, cnt))
        d += 1 if d == 2 else 2
    if x > 1:
        factors.append((x, 1))
    return factors

def count_in_factorial(n, q):
    cnt = 0
    while n:
        n //= q
        cnt += n
    return cnt

def min_n_for_factor(q, e):
    lo, hi = 1, q * e
    while lo < hi:
        mid = (lo + hi) // 2
        if count_in_factorial(mid, q) >= e:
            hi = mid
        else:
            lo = mid + 1
    return lo

def solve():
    p = II()
    if p == 1:
        print(1)
        return
    ans = 1
    for q, e in prime_factors(p):
        ans = fmax(ans, min_n_for_factor(q, e))
    print(ans)

# t = 1
t = II()
for _ in range(t):
    solve()
