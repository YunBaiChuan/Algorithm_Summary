**思路：思维 + 滑动窗口。题目要算的是全为0，或者全为1，其中可穿插2的最长连续字串长度。因此我们可以写一个滑动窗口的函数，各自算一遍0/2以及1/2的情况，其中的较大值为答案**

**Python代码：**
```python
import sys
from collections import Counter

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

def cal(s, tg):
    ans = 1 if tg in s else 0
    l = 0
    cnt = Counter()
    for r, x in enumerate(s):
        cnt[x] += 1
        while cnt[tg] >= 1:
            cnt[s[l]] -= 1
            if cnt[s[l]] == 0:
                del cnt[s[l]]
            l += 1
        ans = fmax(ans, r - l + 1)
    return ans

def solve():
    n = II()
    s = I()
    print(fmax(cal(s, '0'), cal(s, '1')))

# t = 1
t = II()
for _ in range(t):
    solve()
```
