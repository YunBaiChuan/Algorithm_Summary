## 第一题：最小稳定下标 I https://leetcode.cn/problems/smallest-stable-index-i/?envType=daily-question&envId=2026-09-04

**思路：前后缀分解 + 模拟。根据题意进行前后缀分解，然后模拟一遍即可**

**Python代码：**
```python
class Solution:
    def firstStableIndex(self, nums: list[int], k: int) -> int:
        n = len(nums)
        inf = 10 ** 18 
        
        mx = [0] * (n + 1)
        for i, x in enumerate(nums):
            mx[i + 1] = max(mx[i], x)
        
        mn = [inf] * (n + 1)
        for i in range(n - 1, -1, -1):
            mn[i] = min(mn[i + 1], nums[i])
        
        for i in range(n):
            if mx[i + 1] - mn[i] <= k:
                return i
        
        return -1
```

**Cpp代码：**
```cpp
class Solution {
public:
    int firstStableIndex(vector<int>& nums, int k) {
        int inf = 1e9;
        int n = nums.size();

        vector<int> mx(n + 1, 0);
        for (int i = 0; i < n; i++) {
            mx[i + 1] = max(mx[i], nums[i]);
        }

        vector<int> mn(n + 1, inf);
        for (int i = n - 1; i >= 0; i--) {
            mn[i] = min(mn[i + 1], nums[i]);
        }

        for (int i = 0; i < n; i++) {
            if (mx[i + 1] - mn[i] <= k) {
                return i;
            }
        }

        return -1;
    }
};
```
