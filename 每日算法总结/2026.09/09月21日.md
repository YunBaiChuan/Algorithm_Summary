def solve():
    n, m, k, seed = LII()

    def rnd():
        nonlocal seed
        p = (1 << 32)
        ret = seed
        seed = (seed ^ ((seed << 13) % p)) % p
        seed = (seed ^ ((seed >> 17) % p)) % p
        seed = (seed ^ ((seed << 5) % p)) % p
        return ret

    res = [[0] * m for _ in range(n)]
    col = [list(range(n)) for _ in range(m)]

    for _ in range(k):
        op = (rnd() % 2) + 1
        if op == 1:
            j = rnd() % m
            x = (rnd() % (n * m)) + 1
            tmp = col[j]
            while tmp:
                i = tmp.pop()
                res[i][j] = x
        else:
            i = rnd() % n
            j = rnd() % m
            res[i][j] = 0
            col[j].append(i)

    ans = 0
    for i in range(n):
        for j in range(m):
            ans ^= res[i][j] * (i * m + j + 1)
    print(ans)
