**思路：二维前缀和。先预处理二维前缀和，之后再$O(mn)$遍历一遍，查一下$k * k$子矩阵是否为1，是的话就加入答案，最终输出结果即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    n, m, k = LII()
    g = [list(map(int, list(I()))) for _ in range(n)]

    pre = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(n):
        row_sum = 0
        pi, gi = pre[i], pre[i + 1]
        for j in range(m):
            row_sum += g[i][j]
            gi[j + 1] = pi[j + 1] + row_sum

    ans = 0
    for i in range(k - 1, n):
        for j in range(k - 1, m):
            s = pre[i + 1][j + 1] - pre[i + 1 - k][j + 1] \
              - pre[i + 1][j + 1 - k] + pre[i + 1 - k][j + 1 - k]
            if s:
                ans += 1
    print(ans)

t = 1
# t = II()
for _ in range(t):
    solve()
```
