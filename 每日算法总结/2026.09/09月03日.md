## 第一题：最大FST距离 https://www.nowcoder.com/practice/6295f81acd1b4fb59c8beed92577f64b?channelPut=tracker2

**思路：数学 + 绝对值。主要是推式子，考虑$|x - y|$，其最大值其实就是$fmax(x - y, y - x)$。那对于本题来说，式子其实就有四种情况，综合一下得到两种情况为：1.$(A_i^2 + i^2) - (A_j^2 + j^2)$；2.$(A_i^2 - i^2) - (A_j^2 - j^2)$，那我们就算四个变量，即令$u = A_i^2 + i^2$、$v = A_i^2 - i^2$，算他们各自的最大及最小值。最终答案就是$fmax(max_u - min_u,\ max_v - min_v)$**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n = II()
    a = [0] + LII()

    min_u = min_v = inf
    max_u = max_v = -inf
    for i in range(1, n + 1):
        u = a[i] * a[i] + i * i
        v = a[i] * a[i] - i * i
        min_u = fmin(min_u, u)
        min_v = fmin(min_v, v)
        max_u = fmax(max_u, u)
        max_v = fmax(max_v, v)

    print(fmax(max_u - min_u, max_v - min_v))

t = 1
# t = II()
for _ in range(t):
    solve()
```

**Cpp代码：都用ll防溢出**
```cpp
#include <iostream>
#include <vector>
using namespace std;

using ll = long long;
const ll inf = 1e18;

void solve() {
    int n;
    cin >> n;
    vector<ll> a(n + 1);
    for (ll i = 1; i < n + 1; i++) {
        cin >> a[i];
    }

    ll min_u = inf, min_v = inf;
    ll max_u = -inf, max_v = -inf;
    for (ll i = 1; i < n + 1; i++) {
        ll u = a[i] * a[i] + i * i;
        ll v = a[i] * a[i] - i * i;
        min_u = min(min_u, u);
        min_v = min(min_v, v);
        max_u = max(max_u, u);
        max_v = max(max_v, v);
    }

    cout << max(max_u - min_u, max_v - min_v) << '\n';
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T = 1;
    // cin >> T;
    while (T--) {
        solve();
    }
}
```
