**思路：贪心 + 模拟。从大块到小块进行贪心的模拟填充，最后判断下需要的s够不够用即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

def solve():
    s = II()
    a, b, c, d, e, f = LII()

    s -= f
    f = 0

    s -= e
    a -= 11 * e
    if a < 0: a = 0
    e = 0

    s -= d
    use_b = min(b, 5 * d)
    b -= use_b
    a -= (20 * d - 4 * use_b)
    if a < 0: a = 0
    d = 0

    if c % 4 == 0:
        s -= c // 4
    else:
        s -= (c + 3) // 4
        tmp = c % 4
        if tmp == 3:
            use_b = min(b, 1)
            b -= use_b
            a -= (9 - 4 * use_b)
            if a < 0: a = 0
        elif tmp == 2:
            use_b = min(b, 3)
            b -= use_b
            a -= (18 - 4 * use_b)
            if a < 0: a = 0
        else:
            use_b = min(b, 5)
            b -= use_b
            a -= (27 - 4 * use_b)
            if a < 0: a = 0
    c = 0

    if b % 9 == 0:
        s -= b // 9
    else:
        s -= (b + 8) // 9
        tmp = b % 9
        a -= 36 - 4 * tmp
        if a < 0: a = 0
    b = 0

    s -= (a + 35) // 36
    a = 0

    if s >= 0:
        print("Yes")
    else:
        print("No")

# t = 1
t = II()
for _ in range(t):
    solve()
```
