**思路：思维。首先用find找到最左侧的'1'，特判一下没找到说明全'0'，直接输出0即可；然后我们就算每一段'1000...0'的元素个数，并直接更新答案即可**

**Python代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y

mod = 10 ** 9 + 7

def solve():
    n = II()
    s = I()

    bg = s.find('1')
    if bg == -1:
        print(0)
        return

    ans = 1
    cnt = 1
    for i in range(bg, n):
        if s[i] == '0':
            cnt += 1
            continue
        ans = ans * cnt % mod
        cnt = 1

    print(ans)

t = 1
# t = II()
for _ in range(t):
    solve()
```
