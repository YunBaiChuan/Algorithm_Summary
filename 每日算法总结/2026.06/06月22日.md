**思路：GCD的LogTrick。这个和位运算里面的LogTrick类似，简单说来就是利用GCD的单调性，即只减少/不变，最终用一个if将$O(n^2)$优化成$O(nlogn)$**

**代码：**
```python
import sys
from array import array

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

class FastIO:
    def __init__(self):
        self.b = b''
        self.i = 0

    def c(self):
        if self.i >= len(self.b):
            self.b = sys.stdin.buffer.read(1 << 16)
            self.i = 0
            if not self.b:
                return 0
        x = self.b[self.i]
        self.i += 1
        return x

    def n(self):
        c = self.c()
        while c <= 32:
            c = self.c()
        x = 0
        while c > 32:
            x = x * 10 + c - 48
            c = self.c()
        return x

def solve():
    f = FastIO()
    n = f.n()
    m = f.n()

    a = array('I', [0]) * (n + 1)
    for i in range(1, n + 1):
        a[i] = f.n()

    h = array('i', [-1]) * (n + 1)
    ne = array('i', [0]) * m
    ql = array('I', [0]) * m

    for i in range(m):
        l = f.n()
        r = f.n()
        ql[i] = l
        ne[i] = h[r]
        h[r] = i

    ans = array('H', [0]) * m
    pre = []
    g = math.gcd

    for r in range(1, n + 1):
        x = a[r]
        cur = []
        for v, e in pre:
            y = g(v, x)
            if cur and cur[-1][0] == y:
                cur[-1] = (y, e)
            else:
                cur.append((y, e))
        if cur and cur[-1][0] == x:
            cur[-1] = (x, r)
        else:
            cur.append((x, r))
        pre = cur

        k = h[r]
        while k != -1:
            l = ql[k]
            cnt = 0
            for _, e in pre:
                if e >= l:
                    cnt += 1
            ans[k] = cnt
            k = ne[k]

    sys.stdout.write('\n'.join(map(str, ans)))

t = 1
# t = II()
for _ in range(t):
    solve()
```
