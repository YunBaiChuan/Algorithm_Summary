**思路：利用bfs预处理每个数值的最短路径/最少操作数，由题意可知前三个操作就是bfs权重，分别为+-1，+-10，+-100；后两个操作中的第一个操作没用，因为初始就是10，第二个操作300就只需要一次操作即可，可以把后两个直接入队列，最终得到整个范围上的最少操作次数，然后累加对应的操作数即可，因为每个位置独立**

**代码：**
```python
import sys
from collections import deque

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y

dirs = [1, -1, 10, -10, 100, -100]
dist = [-1] * 301
dist[10] = 0
dist[300] = 1

q = deque()
q.append(10)
q.append(300)
while q:
    for _ in range(len(q)):
        x = q.popleft()
        for dx in dirs:
            nx = x + dx
            if 10 <= nx <= 300 and dist[nx] == -1:
                dist[nx] = dist[x] + 1
                q.append(nx)

def solve():
    arr = LII()
    cnt = 0
    for x in arr:
        cnt += dist[x]
    print(cnt)

# t = 1
t = II()
for _ in range(t):
    solve()
```
