## 第一题：和谐并处的车 https://www.nowcoder.com/practice/1a44f9b7eee348f9ba8233a551e48ca6?channelPut=tracker2

**思路：思维 + 并查集。首先我们要让所有车归位，并且初始状态下，每辆车在不同行、不同列，因此移动是不会有车阻挡的，我们只会在不同行的列上进行移动；那最少移动次数该如何计算？其实分成两个部分：1.base，基础移动，每辆车如果x != y，他肯定要进行一次移动，才能到目标位置；2.extra，额外移动，每辆车移动位置已经有车了，就需要先把起冲突的车先放到不相关的位置，等其他车移动好了，再放回正确位置（题目保证了一定存在一个可以放的无关位置）**

**base移动好算，主要是extra移动如何计算。注意到，我们要进行额外移动的时候，就是其形成了环，比如2 -> 1，1 -> 2这个移动，他就形成了环。要破环这个环，其实就是前面说的，额外将环中任意一个元素放到一个可以放的无关位置，也就是操作一次，一个环。因此最终答案就是base + 不同环数量**

class UnionFind:
    def __init__(self, n):
        self.fa = list(range(n))
        self.size = [1] * n
        self.sets = n
        self.has_cycle = [False] * n

    def find(self, x):
        if x != self.fa[x]:
            self.fa[x] = self.find(self.fa[x])
        return self.fa[x]

    def is_same(self, x, y):
        return self.find(x) == self.find(y)

    def union(self, x, y):
        fx, fy = self.find(x), self.find(y)
        if fx == fy:
            self.has_cycle[fx] = True
            return False
        if self.size[fx] < self.size[fy]:
            fx, fy = fy, fx
        self.fa[fy] = fx
        self.size[fx] += self.size[fy]
        self.sets -= 1
        if self.has_cycle[fy]:
            self.has_cycle[fx] = True
        return True

def solve():
    n, m = MII()
    uf = UnionFind(n + 5)

    not_on_diag = 0
    used_node = set()

    for _ in range(m):
        x, y = MII()
        if x == y:
            continue
        not_on_diag += 1
        uf.union(x, y)
        used_node.add(x)
        used_node.add(y)

    cycles = 0
    for node in used_node:
        root = uf.find(node)
        if uf.find(root) == root and uf.has_cycle[root]:
            cycles += 1
            uf.has_cycle[root] = False

    print(not_on_diag + cycles)

# t = 1
t = II()
for _ in range(t):
    solve()
