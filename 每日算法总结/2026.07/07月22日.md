**思路：思维 + 并查集。首先要求最少的染色次数，其实就是连通块的数目，每个连通块中的所有点，只需要再染一次色就可以到达非标记节点；其次就是染色方式，这个和连通块与它周围有几条非染色边有关，我们可以算出每个连通块周围有几条非染色边，然后将他们相乘并取mod，就可以得到方案数**

mod = 10 ** 9 + 7

class UnionFind:
    def __init__(self, n):
        self.fa = list(range(n))
        self.size = [1] * n
        self.sets = n

    def find(self, x):
        root = x
        while root != self.fa[root]:
            root = self.fa[root]
        while x != root:
            nxt = self.fa[x]
            self.fa[x] = root
            x = nxt
        return root

    def is_same(self, x, y):
        fx, fy = self.find(x), self.find(y)
        return fx == fy

    def union(self, x, y):
        fx, fy = self.find(x), self.find(y)
        if fx == fy:
            return False
        self.fa[fx] = fy
        self.size[fy] += self.size[fx]
        self.sets -= 1
        return True

def solve():
    n, k = MII()
    a = LII()
    marked = set(a)

    edges = []
    for _ in range(n - 1):
        u, v = MII()
        edges.append((u, v))

    uf = UnionFind(n + 1)
    for u, v in edges:
        if u in marked and v in marked:
            uf.union(u, v)

    cnt = defaultdict(int)
    for u, v in edges:
        fu = u in marked
        fv = v in marked
        if fu and not fv:
            cnt[uf.find(u)] += 1
        elif not fu and fv:
            cnt[uf.find(v)] += 1

    ans_cost = len(cnt)
    ans_way = 1
    for v in cnt.values():
        ans_way = ans_way * v % mod

    print(ans_cost, ans_way)

t = 1

for _ in range(t):
    solve()
