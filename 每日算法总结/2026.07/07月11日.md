## 第一题：统计完全连通分量的数量 https://leetcode.cn/problems/count-the-number-of-complete-components/description/?envType=daily-question&envId=2026-07-11

**思路：并查集 + 连通块判断。首先用并查集将元素划分成连通块，接着对于每个连通块，判断其度数是否为其连通块大小 - 1，如果是的话，说明为完全连通分量，否则就不是。最终，输出答案即可**

**代码：**
```python
class UnionFind:
    def __init__(self, n):
        self.fa = list(range(n))
        self.size = [1] * n
        self.sets = n

    def find(self, x):
        if x != self.fa[x]:
            self.fa[x] = self.find(self.fa[x])
        return self.fa[x]

    def is_same(self, x, y):
        fx, fy = self.find(x), self.find(y)
        return fx == fy

    def union(self, x, y):
        fx, fy = self.find(x), self.find(y)
        if fx == fy:
            return False
        self.fa[fx] = fy
        self.size[fy] += self.size[fx]
        self.sets -= 1
        return True

class Solution:
    def countCompleteComponents(self, n: int, edges: List[List[int]]) -> int:
        uf = UnionFind(n)
        deg = [0] * n 
        for u, v in edges:
            deg[u] += 1
            deg[v] += 1
            if not uf.is_same(u, v):
                uf.union(u, v)
        
        groups = [[] for _ in range(n)]
        for i in range(n):
            groups[uf.find(i)].append(i)
        
        ans = 0
        for row in groups:
            if row:
                target = len(row) - 1
                flag = True 
                for x in row:
                    if deg[x] != target:
                        flag = False 
                        break 
                ans += 1 if flag else 0 
        return ans 
```
