**思路：数学。有个数学性质：对于能够被4整除的数来说，只需要最后两位能够被4整除即可。因此我们就枚举可以组成的两位数，判断其能否被4整除即可，如果可以的话，就把它放在末尾，输出YES；如果都不行，输出NO**

def solve():
    sn = I()
    if int(sn) % 4 == 0:
        print("YES")
        return

    for i in range(len(sn)):
        for j in range(len(sn)):
            if i == j: continue
            tmp = 10 * int(sn[i]) + int(sn[j])
            if tmp % 4 == 0:
                print("YES")
                return

    print("NO")

# t = 1
