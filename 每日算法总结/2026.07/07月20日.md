## 第一题：小红的双生英雄 https://www.nowcoder.com/practice/bb8efeec4e39421d8fe6516169c98083?channelPut=tracker2

**思路：分组背包。主要是先通过预处理得到单英雄/双生英雄对应的分组，对于单英雄来说就是它本身；对于双生英雄来说，可以是其中的某一个英雄，也可以是两个英雄都选并获得增益。处理得到分组之后，再跑一个分组背包，算其中的最大值即可**

**代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, C, m = LII()

    a = [0] * n
    b = [0] * n
    for i in range(n):
        a[i], b[i] = LII()

    pair_id = [-1] * n
    pair_info = []
    for _ in range(m):
        pid = len(pair_info)
        u, v, w = LII()
        u -= 1
        v -= 1
        pair_id[u] = pid
        pair_id[v] = pid
        pair_info.append((u, v, w))

    groups = []
    used_pair = [False] * m

    for i in range(n):
        pid = pair_id[i]
        if pid == -1:
            groups.append([(1, a[i], b[i])])
        else:
            if not used_pair[pid]:
                used_pair[pid] = True
                u, v, w = pair_info[pid]
                opts = []
                opts.append((1, a[u], b[u]))
                opts.append((1, a[v], b[v]))
                opts.append((2, a[u] + a[v], b[u] + b[v] + w))
                groups.append(opts)

    dp = [[-inf] * (C + 1) for _ in range(5)]
    dp[0][0] = 0

    for opts in groups:
        new_dp = [row[:] for row in dp]
        for num, cost, val in opts:
            if cost > C:
                continue
            for pre in range(5 - num):
                for c in range(C - cost + 1):
                    if dp[pre][c] != -inf:
                        new_dp[pre + num][c + cost] = fmax(
                            new_dp[pre + num][c + cost],
                            dp[pre][c] + val
                        )
        dp = new_dp

    ans = 0
    for i in range(5):
        for c in range(C + 1):
            ans = fmax(ans, dp[i][c])
    print(ans)

t = 1
# t = II()
for _ in range(t):
    solve()
```
