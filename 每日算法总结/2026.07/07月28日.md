**思路：主席树（可持久化权值线段树）+ 离散化。这题是区间中位数查询，本质是区间第 k 小问题（k = 区间长度/2 + 1）。比较复杂，先放着**

import sys
from bisect import bisect_left

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

    return map(int, input().split())

    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

class PersistentSegmentTree:
    def __init__(self, n, m):
        """
        n: 原数组长度
        m: 离散化后的值域大小
        """
        self.n = n
        self.m = m
        self.size = n * 25
        self.idx = 0

        self.lc = [0] * self.size
        self.rc = [0] * self.size
        self.sumv = [0] * self.size
        self.root = [0] * (n + 1)

    def _update(self, prev, l, r, x):
        """递归更新，返回新节点索引"""
        self.idx += 1
        p = self.idx

        self.lc[p] = self.lc[prev]
        self.rc[p] = self.rc[prev]
        self.sumv[p] = self.sumv[prev] + 1

        if l != r:
            mid = (l + r) >> 1
            if x <= mid:
                self.lc[p] = self._update(self.lc[prev], l, mid, x)
            else:
                self.rc[p] = self._update(self.rc[prev], mid + 1, r, x)
        return p

    def build(self, arr):
        """根据数组构建主席树"""
        for i in range(1, self.n + 1):
            self.root[i] = self._update(self.root[i - 1], 1, self.m, arr[i - 1])

    def query_kth(self, left_root, right_root, k):
        """查询区间 [left_root+1, right_root] 的第 k 小值（返回离散化后的值）"""
        l, r = 1, self.m
        u, v = self.root[left_root], self.root[right_root]

        while l < r:
            mid = (l + r) >> 1
            left_cnt = self.sumv[self.lc[v]] - self.sumv[self.lc[u]]

            if left_cnt >= k:
                u = self.lc[u]
                v = self.lc[v]
                r = mid
            else:
                k -= left_cnt
                u = self.rc[u]
                v = self.rc[v]
                l = mid + 1

        return l

def solve():
    n, q = MII()
    a = LII()

    b = sorted(set(a))
    m = len(b)
    a = [bisect_left(b, x) + 1 for x in a]

    pst = PersistentSegmentTree(n, m)
    pst.build(a)

    out = []
    for _ in range(q):
        l, r = MII()
        kth = (r - l + 1) // 2 + 1
        ans_val = pst.query_kth(l - 1, r, kth)
        out.append(str(b[ans_val - 1]))

    print('\n'.join(out))

t = 1
# t = II()
for _ in range(t):
    solve()
