**思路：Kosaraju 算法（迭代版）+ 拓扑排序。通过判断缩点后的 DAG 是否是一条单链来判断是否可达**

**代码：**

import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

def solve():
    n, m = MII()
    g = [[] for _ in range(n + 1)]
    rg = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = MII()
        g[u].append(v)
        rg[v].append(u)

    vis = [0] * (n + 1)
    order = []
    for i in range(1, n + 1):
        if vis[i]:
            continue
        st = [(i, 0)]
        while st:
            u, ph = st.pop()
            if ph == 0:
                if vis[u]:
                    continue
                vis[u] = 1
                st.append((u, 1))
                for v in g[u]:
                    if not vis[v]:
                        st.append((v, 0))
            else:
                order.append(u)

    comp = [0] * (n + 1)
    cid = 0
    for u in reversed(order):
        if comp[u]:
            continue
        cid += 1
        comp[u] = cid
        st = [u]
        while st:
            x = st.pop()
            for v in rg[x]:
                if not comp[v]:
                    comp[v] = cid
                    st.append(v)

    if cid == 1:
        print("Yes")
        return

    indeg = [0] * (cid + 1)
    dag = [[] for _ in range(cid + 1)]
    seen = set()
    for u in range(1, n + 1):
        cu = comp[u]
        for v in g[u]:
            cv = comp[v]
            if cu != cv:
                key = cu * (cid + 1) + cv
                if key not in seen:
                    seen.add(key)
                    dag[cu].append(cv)
                    indeg[cv] += 1

    q = [i for i in range(1, cid + 1) if indeg[i] == 0]
    h = 0
    ok = True
    while h < len(q):
        if len(q) - h > 1:
            ok = False
            break
        u = q[h]
        h += 1
        for v in dag[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    print("Yes" if ok and h == cid else "No")

t = 1
# t = II()
for _ in range(t):
    solve()
