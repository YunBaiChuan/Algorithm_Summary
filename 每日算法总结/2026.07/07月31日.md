**思路：思维。题目大意是说给定$[1x], [2x], [3x]... [nx]$的整数数组，求其中的x上界。那其实有个要求就是$i * x < a[i] + 1.0$，即$x < (a[i] + 1.0) / i$，我们遍历一遍，取其中的最小值就是答案了**

**代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n = II()
    a = LII()

    ans = inf
    for i, x in enumerate(a, 1):
        ans = fmin(ans, (x + 1.0) / i)
    print(f"{ans:.10f}")

t = 1
# t = II()
for _ in range(t):
    solve()
```
