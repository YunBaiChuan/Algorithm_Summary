**思路：思维。如果直接按照题目进行模拟的话，时间复杂度为$O(nq)$，肯定会超时。考虑到题目只有去重这一操作，我们可以首先预处理出超过1个数的字符串，这是我们后续有操作2时，必然会删除的字符串，记作ans；然后我们再维护一个a的集合，方便后续新增别的字符串时能够正常算出需要删除的次数；接着就根据操作来更新ans，如果新增字符串是有的，那么ans += 1；否则就将这个字符串添加到a的集合中。最终，收集到所有ans，直接输出即可**

**代码：**
```python
import sys
from collections import Counter

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, q = MII()
    a = [I() for _ in range(n)]
    sa = set(a)

    cnt = Counter(a)
    ans = 0
    for k, v in cnt.items():
        ans += v - 1

    out = []
    for _ in range(q):
        op, *b = LI()
        if int(op) == 1:
            s = b[0]
            if s in sa:
                ans += 1
            else:
                sa.add(s)
        else:
            out.append(str(ans))
            ans = 0

    print("\n".join(out))

t = 1
# t = II()
for _ in range(t):
    solve()
```
