## 第一题：最大的得分的路径数目 https://leetcode.cn/problems/number-of-paths-with-max-score/?envType=daily-question&envId=2026-07-05

**思路：网格图dp。主要是怎么想到统计最大得分路径的数量，这其实可以通过额外创建一个dp_count数组进行统计。具体来说就是，在计算基础的dp_sum时，根据题意描述的三个方向进行dp_sum == mx的判断，如果相同就累加到此时的路径总数total中，然后同时更新两个dp数组。最终，取mod输出即可**

**代码：**
```python
class Solution:
    def pathsWithMaxScore(self, board: List[str]) -> List[int]:
        MOD = 10 ** 9 + 7
        m, n = len(board), len(board[0])
        # 把board转化为数字，方便后面处理
        num_board = [[0] * n for _ in range(m)]
        for i in range(m):
            for j in range(n):
                c = board[i][j]
                if c == 'E' or c == 'S':
                    num_board[i][j] = 0
                elif c == 'X':
                    num_board[i][j] = -1
                else:
                    num_board[i][j] = int(c)

        # 动态规划dp，直接写递推
        dp_sum = [[-inf] * (n + 1) for _ in range(m + 1)] # 求最大代价
        dp_count = [[0] * (n + 1) for _ in range(m + 1)] # 求方案数
        # 初始化
        dp_sum[1][1] = 0
        dp_count[1][1] = 1
        for i in range(1, m + 1):
            for j in range(1, n + 1):
                if num_board[i - 1][j - 1] == -1: # 如果当前位置是障碍物，说明无法到达
                    continue
                # 最大代价
                max_sum = max(dp_sum[i - 1][j], dp_sum[i][j - 1], dp_sum[i - 1][j - 1])
                if max_sum == -inf:  
                    continue
                # 路径计算
                total = 0
                if dp_sum[i - 1][j] == max_sum: # 左方
                    total += dp_count[i - 1][j]
                if dp_sum[i][j - 1] == max_sum: # 上方
                    total += dp_count[i][j - 1]
                if dp_sum[i - 1][j - 1] == max_sum: # 左上方
                    total += dp_count[i - 1][j - 1]
                # 状态转移
                dp_sum[i][j] = max_sum + num_board[i - 1][j - 1]
                dp_count[i][j] = total % MOD
        
        if dp_sum[m][n] == -inf:
            return [0, 0]
        return [dp_sum[m][n] % MOD, dp_count[m][n] % MOD]
```

## 第二题：牛牛的方格图 https://www.nowcoder.com/practice/17efc53f90664c18b3f9230dc456530b?channelPut=tracker2

**思路：思维 + 二维差分。这个不好想到二维差分，要熟练度足够高才行。从题意可知，对于每个颜色对来说，我们要将他们视为矩形的对角线，然后这个矩阵进行染色。这个其实和二维差分是一样的，为什么？二维差分也是需要对角线的两个点进行差分，然后再进行一次二维前缀和就可以把这个矩形染色。因此，我们就先统计出整个矩阵中，每个颜色的4个参数，即mnx，mxx，mny，mxy。如果这个颜色有超过两个，就进行二维差分，最终算一遍二维前缀和，统计下没有被染色的个数即可**

**代码：**

import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

MAXI = 10 ** 6

def solve():
    n, m = MII()
    a = [LII() for _ in range(n)]

    minx = [inf] * (MAXI + 1)
    maxx = [-inf] * (MAXI + 1)
    miny = [inf] * (MAXI + 1)
    maxy = [-inf] * (MAXI + 1)

    for i in range(n):
        for j in range(m):
            c = a[i][j]
            if i < minx[c]:
                minx[c] = i
            if i > maxx[c]:
                maxx[c] = i
            if j < miny[c]:
                miny[c] = j
            if j > maxy[c]:
                maxy[c] = j

    diff = [[0] * (m + 1) for _ in range(n + 1)]
    for c in range(1, MAXI + 1):
        u = minx[c]
        d = maxx[c]
        l = miny[c]
        r = maxy[c]
        if u != inf and (u < d or l < r):
            d += 1
            r += 1
            diff[u][l] += 1
            diff[u][r] -= 1
            diff[d][l] -= 1
            diff[d][r] += 1

    for i in range(n):
        for j in range(m):
            diff[i][j + 1] += diff[i][j]

    res = 0
    for i in range(n):
        for j in range(m):
            if diff[i][j] == 0:
                res += 1
            diff[i + 1][j] += diff[i][j]
    print(res)

t = 1
# t = II()
for _ in range(t):
    solve()
