### 1.小苯的01背包 https://www.nowcoder.com/practice/32407155397c443a95f895f11942f02f?channelPut=tracker2

**思路：这题不能用01背包板子，因为&没有+那样的单调性，状态转移写不出来。因此，本题需要用位枚举，从高位到低位进行枚举，然后判断能够达到答案**

**代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, k = MII()
    a = [LII() for _ in range(n)]

    ans = 0
    for mask in range(10, -1, -1):
        target = ans | (1 << mask)
        v_and = (1 << 11) - 1
        ok = False

        for i in range(n):
            v, w = a[i]
            if w & target == target: # 能选的都选，可以把体积变小，并且当前位的结果依然最优
                v_and &= v
                ok = True

        if ok and v_and <= k:
            ans = target

    print(ans)

t = 1
# t = II()
for _ in range(t):
    solve()
```

### 2.统计节点和为偶数的连通子图 https://leetcode.cn/problems/count-connected-subgraphs-with-even-node-sum/description/

**思路：位运算枚举子集 + dfs判断连通性。这个要进行局部的枚举子集的连通性判断，不能用并查集全局连通性判断，对于本题来说，用dfs判断连通性写起来比较简单**

**代码：**
```python
class Solution:
    def evenSumSubgraphs(self, nums: list[int], edges: list[list[int]]) -> int:
        n = len(nums)
        g = [[] for _ in range(n)]
        for x, y in edges:
            g[x].append(y)
            g[y].append(x)

        u = (1 << n) - 1
        ans = 0

        for sub in range(1, u + 1):
            xor_sum = 0
            for i, x in enumerate(nums):
                if sub >> i & 1:
                    xor_sum ^= x
            if xor_sum:
                continue

            def dfs(x: int) -> None:
                nonlocal vis
                vis |= 1 << x
                for y in g[x]:
                    if (vis >> y & 1) == 0:
                        dfs(y)

            vis = u ^ sub
            dfs(sub.bit_length() - 1)
            if vis == u:
                ans += 1

        return ans
```

### 3.出题需要加法 https://ac.nowcoder.com/acm/contest/134527/C

**思路：位运算。首先由题意可知，满足条件的数要么是二进制中有两个1，要么是二进制中有一个1，并且值不为1；所以说满足条件的数是很少的，我们就可以提前预处理出所有满足条件的数，并且将这些数进行排序；最后，利用二分查找优化，算出$[l, r]$区间中，有多少个满足条件的数**

```python
'''

'''

ans = set()
cur = 0
for i in range(64):
    cur |= 1 << i
    pre = cur
```

### 4.或与异或 https://www.nowcoder.com/practice/a4bddff248c1424c90ee6cf309cf303f?channelPut=tracker2

**思路：位运算 + 思维。根据或、与、异或这三个操作，虽然可以进行无限次操作，但是结果是有限的。因此，我们可以算出一个ans集合，然后判断target是否在ans集合中即可。这里面有个恒等式为(a & b) ^ b = ~a & b需要记一下**

**代码：**
```python

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    a, b, target = LII()
    ans = {0, a, b, a | b, a & b, a ^ b, a & ~b, ~a & b}
    print("YES" if target in ans else "NO")

# t = 1
t = II()
for _ in range(t):
    solve()
```

### 5.医生 https://www.nowcoder.com/practice/91875c1230ea4e8eb7ae4f3338e5be46?channelPut=tracker2

**思路：位运算。首先发现k比较小，因此可以用位运算预处理出所有药物组合可以覆盖的症状，然后对每个病人枚举预处理的药物组合，判断其能否覆盖病症，或者说病症是否为这个药物组合的子集。最终，输出结果即可**

**代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

def solve():
    n, m = MII()

    patients = []
    for _ in range(n):
        s = I()
        patients.append(int(s, 2))

    k = II()
    drugs = []
    for _ in range(k):
        s = I()
        drugs.append(int(s, 2))

    total = 1 << k

    cover = [0] * total
    cnt = [0] * total

    for mask in range(total):
        cur = 0
        num = 0
        for j in range(k):
            if (mask >> j) & 1:
                num += 1
                cur |= drugs[j]
        cover[mask] = cur
        cnt[mask] = num

    for patient in patients:
        ans = inf
        for mask in range(total):
            if (cover[mask] | patient) == cover[mask]:
                ans = min(ans, cnt[mask])
        print(-1 if ans == inf else ans)

t = 1
# t = II()
for _ in range(t):
    solve()
```

### 6.不同异或三元组的数目I https://leetcode.cn/problems/number-of-unique-xor-triplets-i/description/?envType=daily-question&envId=2026-07-23

**思路：思维 + 位运算。从题意可知，元素是1-n的排列，并且三元组异或元素可以任意选择。可以发现有以下规律：1.n <= 2时，答案就是其本身；2.n > 2时，三元组异或的种类由最高比特位决定，把玩样例可知，最高比特位的更低位都可以通过前面两个元素的异或构造出来，最终再异或上最高比特位，就可以获得新的异或和，因此答案为1 << max_length**

**代码：**
```python
class Solution:
    def uniqueXorTriplets(self, nums: List[int]) -> int:
        n = len(nums)
        if n == 1 or n == 2:
            return n
        mx_length = n.bit_length()
        return (1 << mx_length)
```

### 7.2^20 https://www.nowcoder.com/practice/c44464a6d8dd42d0bb09c76800d20116?channelPut=tracker2

**思路：思维 + 位运算。考虑题目给定的两种操作：1.直接+1；2.不断*2，其实就是左移一位。题目要求通过尽量少的这两种操作，使得最终得到的数能够被$2^{20}$整除。这里其实要知道一个结论，对于能够被$2^x$整除的数来说，他的末尾x位都为0。以能够被$2^2$整除为例，那么满足这个条件的数，末尾两位应该都为0才行。所以说本题就是要让后20位都为0，并且尽量要让操作少，那就应该先进行+1操作，然后最多进行20次左移操作。不断算操作中的最小值，得到最终答案**

**代码：**
```python
import sys
input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

init = 1 << 20

def solve():
    n = II()
    ans = 20 # 最差情况为乘20次直接得到2^20
    for i in range(20): # 枚举优化，最多操作都是20了，i操作肯定 < 20
        tmp = n + i # 先加再乘，优先保证右侧零尽可能多，则操作会尽可能少
        j = 0
        while j < 20:
            if (tmp << j) % init == 0:
                ans = fmin(ans, i + j)
            j += 1
    print(ans)

# t = 1
t = II()
for _ in range(t):
    solve()
```

## LogTrick

### 1.小辰刚学gcd https://www.nowcoder.com/practice/3472cf2e23534a3896f2112f11ab4aac?channelPut=tracker2

**思路：GCD的LogTrick。这个和位运算里面的LogTrick类似，简单说来就是利用GCD的单调性，即只减少/不变，最终用一个if将$O(n^2)$优化成$O(nlogn)$**

**代码：**
```python
import sys
from array import array

input = lambda: sys.stdin.readline().strip()

import math
inf = 10 ** 18

def I():
    return input()

def II():
    return int(input())

def MII():
    return map(int, input().split())

def LI():
    return input().split()

def LII():
    return list(map(int, input().split()))

def LFI():
    return list(map(float, input().split()))

fmax = lambda x, y: x if x > y else y
fmin = lambda x, y: x if x < y else y
isqrt = lambda x: int(math.sqrt(x))

'''

'''

class FastIO:
    def __init__(self):
        self.b = b''
        self.i = 0

    def c(self):
        if self.i >= len(self.b):
            self.b = sys.stdin.buffer.read(1 << 16)
            self.i = 0
            if not self.b:
                return 0
        x = self.b[self.i]
        self.i += 1
        return x

    def n(self):
        c = self.c()
        while c <= 32:
            c = self.c()
        x = 0
        while c > 32:
            x = x * 10 + c - 48
            c = self.c()
        return x

def solve():
    f = FastIO()
    n = f.n()
    m = f.n()

    a = array('I', [0]) * (n + 1)
    for i in range(1, n + 1):
        a[i] = f.n()

    h = array('i', [-1]) * (n + 1)
    ne = array('i', [0]) * m
    ql = array('I', [0]) * m

    for i in range(m):
        l = f.n()
        r = f.n()
        ql[i] = l
        ne[i] = h[r]
        h[r] = i

    ans = array('H', [0]) * m
    pre = []
    g = math.gcd

    for r in range(1, n + 1):
        x = a[r]
        cur = []
        for v, e in pre:
            y = g(v, x)
            if cur and cur[-1][0] == y:
                cur[-1] = (y, e)
            else:
                cur.append((y, e))
        if cur and cur[-1][0] == x:
            cur[-1] = (x, r)
        else:
            cur.append((x, r))
        pre = cur

        k = h[r]
        while k != -1:
            l = ql[k]
            cnt = 0
            for _, e in pre:
                if e >= l:
                    cnt += 1
            ans[k] = cnt
            k = ne[k]

    sys.stdout.write('\n'.join(map(str, ans)))

t = 1
# t = II()
for _ in range(t):
    solve()
```
